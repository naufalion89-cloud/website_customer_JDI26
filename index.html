Integrasi Google Apps Script (Code.gs)

Salin kode berikut ke Google Apps Script (Extensions -> Apps Script pada Google Spreadsheet Anda):

function doGet(e) {
  return HtmlService.createHtmlOutput("API PT JALUR DATA INDONESIA Aktif.");
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.openById("1bR4QVxYM1sTtdaNgCS3uNA9vQHuw7iCL2cXu_DDN3TM");
    var sheet = ss.getSheetByName("Pendaftaran") || ss.insertSheet("Pendaftaran");
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "No. Reservasi", "Nama", "No. WA", "Desa", "Paket", "Alamat", "Foto Lokasi", "Status"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.reservationId,
      data.clientName,
      data.clientPhone,
      data.clientVillage,
      data.clientPackage,
      data.clientAddress,
      data.locationPhoto || "Tidak Ada",
      "Menunggu Verifikasi"
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", id: data.reservationId }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
